class Solution {
public:
    bool isVowel(char ch)
    {
        return ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u';
    }

    int countVowelSubstrings(string word) 
    {
        int n = word.size();
        int ans =0;
        for(int i=0;i<n;i++)
        {
            unordered_set<char> s1;
            for(int j=i;j<n;j++)
            {
                if(!isVowel(word[j]))
                {
                    break;
                }
                s1.insert(word[j]);

                if(s1.size() == 5)
                {
                    ans++;
                }
            }
        }
        return ans;
    }
};
