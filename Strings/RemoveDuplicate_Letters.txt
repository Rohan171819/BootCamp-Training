class Solution {
public:
    string removeDuplicateLetters(string s) 
    {
        vector<int> last(26);
        vector<bool> check(26,false);
        string result;
        for(int i=0;i<s.length();i++)
        {   
            char ch = s[i];
            last[ch - 'a'] = i;
        }
        
        for(int i=0;i<s.length();i++)
        { 
            char ch = s[i];
            if(check[ch - 'a'])
            {
                continue;
            }
            
            while(!result.empty() && result.back() > ch && last[result.back()- 'a'] > i)
            {
                check[result.back() - 'a'] = false;
                result.pop_back();
            }
            result.push_back(ch);
            check[ch - 'a'] = true;
        }
        return result;
    }
    
};
