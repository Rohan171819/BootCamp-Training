class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) 
    {
        unordered_map<string,vector<string>> mp;
        for(string st : strs)
        {
            string temp = st;
            sort(temp.begin(),temp.end());
            mp[temp].push_back(st);
        }
        vector<vector<string>> result;
        for(auto x : mp)
        {
            result.push_back(x.second);
        }
        return result;
    }
};
