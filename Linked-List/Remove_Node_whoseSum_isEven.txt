#include<iostream>
#include<list>
using namespace std;
int main()
{
    list<int> l1;
    int n;
    cout<<"Enter size of list:"<<endl;
    cin>>n;
    
    cout<<"Enter values:"<<endl;
    for(int i=0;i<n;i++)
    {
        int x;
        cin>>x;
        l1.push_back(x);
    }
    for(int i : l1)
    {
        int temp = i;
        int sum=0;
        while(temp > 0)
        {
            sum += temp % 10;
            temp = temp /10;
        }
        if(sum % 2 !=0)
        {
            cout<<i<<" ";
        }
    }
    return 0;
}
