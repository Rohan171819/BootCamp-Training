#include<iostream>
using namespace std;
struct Node
{
    int data;
    struct Node *Next;
    
    Node(int val)
    {
        data= val;
        Next = NULL;
    }
};

int main()
{
    int n;
    cout<<"Enter size:"<<endl;
    cin>>n;
    struct Node *Head = NULL;
    struct Node *Tail= NULL;
    cout<<"Enter values:"<<endl;
    for(int i=0;i<n;i++)
    {
        int x;
        cin>>x;
        struct Node *newNode = new Node(x);
        if(Head == NULL)
        {
            Head = newNode;
            Tail = newNode;
        }
        else
        {
            Tail->Next = newNode;
            Tail = newNode;
        }
    }
    
    Tail->Next = Head;
    struct Node *temp = Head;
    do
    {
        cout<<temp->data<<" ";
        temp=temp->Next;
    }
    while(temp!=Head);
}
