#include<iostream>
using namespace std;
struct Node
{
    int data;
    struct Node* prev;
    struct Node* Next;
    
    Node(int val)
    {
        data=val;
        prev=NULL;
        Next=NULL;
    }
};
int main()
{
  int n;
  cin>>n;
  struct Node *head=NULL;
  struct Node *tail=NULL;
  for(int i=0;i<n;i++)
  {
      int value;
      cout<<"Enter value:"<<endl;
      cin>>value;
      struct Node *newNode = new Node(value); 
      if(head==NULL)
      {
          head= newNode;
          tail= newNode;
      }
      else
      {
          tail->Next= newNode;
          newNode->prev=tail;
          tail=newNode;
      }
  }
  struct Node*temp=head;
  while(temp!=NULL && temp->Next!=NULL)
  {
      cout<<temp->data;
      temp=temp->Next;
  }
}
