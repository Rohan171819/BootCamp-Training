#include<iostream>
#include<list>
using namespace std;

int main()
{
    int n;
    cin>>n;
    int x;
    list<int>l1;
    for(int i=0;i<n;i++)
    {
        cin>>x;
        l1.push_back(x);
    }
    for(int a: l1)
    {
        cout<<a<<" ";
    }
    cout<<endl;
    list<int> l2(l1);
    for(int b: l2)
    {
        cout<<b<<" ";
    }
}
