#include<iostream>
#include<list>
using namespace std;

int main()
{
    int n,key,count=0;
    cout<<"Enter size:"<<endl;
    cin>>n;
    list<int> l1;
    cout<<"Enter key:"<<endl;
    cin>>key;
    cout<<"Enter list elements:"<<endl;
    for(int i=0;i<n;i++)
    {
        int x;
        cin>>x;
        l1.push_back(x);
    }

    for(int i: l1)
    {
        if(key == i)
        {
            count++;
        }
    }
    cout<<count;
}
