#include<iostream>
#include<list>
using namespace std;
int main()
{
    list<int> l1;
    list<int> result;
    int n1;
    cout<<"Enter n1 size:"<<endl;
    cin>>n1;
    cout<<"Enter Linked list elements:"<<endl;
    for(int i=0;i<n1;i++)
    {
        int x;
        cin>>x;
        l1.push_back(x);
    }
    
    for(int i : l1)
    {
        result.push_front(i);
    }
    
    for(int i: result)
    {
        cout<<i<<" ";
    }
}
