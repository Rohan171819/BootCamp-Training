#include<iostream>
using namespace std;
struct Node
{
    int data;
    struct Node *Next;

    Node(int val)
    {
        data=val;
        Next=NULL;
    }
};

int main()
{
    int n,count=0;
    cout<<"Enter the number of nodes:"<<endl;
    cin>>n;
    struct Node *head=NULL;
    struct Node *tail = NULL;
    for(int i=0;i<n;i++)
    {
        int value;
        cout<<"Enter the value:"<<endl;
        cin>>value;
        struct Node *newNode = new Node(value);
        count++;
    
        if(head == NULL)
        {
            head = newNode;
            tail = newNode;
        }
        else 
        {
            tail->Next = newNode;
            tail = newNode;
        }
    }
    cout<<"Total Nodes are:"<<count<<endl;
}
