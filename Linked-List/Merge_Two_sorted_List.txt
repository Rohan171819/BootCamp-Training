#include<iostream>
#include<list>
using namespace std;
int main()
{
    list<int> l1;
    list<int> l2;
    list<int> result; 
    int n1,n2;
    cout<<"Enter l1 size:"<<endl;
    cin>>n1;
    cout<<"Enter l2 size:"<<endl;
    cin>>n2;
    cout<<"Enter Sorted elements in linked list1:"<<endl;
    for(int i=0;i<n1;i++)
    {
        int  x;
        cin>> x;
        l1.push_back(x);
    }
    cout<<"Enter Sorted elements in linked list2:"<<endl;
    for(int i=0;i<n2;i++)
    {
        int  x;
        cin>> x;
        l2.push_back(x);
    }
    
    l1.merge(l2);
    l1.sort();
    cout<<"Merged List is:"<<endl;
    for(int x: l1)
    {
        cout<<x<<" ";
    }
}
